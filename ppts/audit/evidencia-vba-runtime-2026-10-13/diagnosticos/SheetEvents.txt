Option Explicit
Public EventCount As Long
Public SelectionCount As Long
Public LastSelection As String
Public SafeWriteMode As Boolean
Public RecursiveProbe As Boolean
Private Sub Worksheet_SelectionChange(ByVal Target As Range)
    SelectionCount = SelectionCount + 1
    LastSelection = Target.Address
End Sub
Private Sub Worksheet_Change(ByVal Target As Range)
    EventCount = EventCount + 1
    If RecursiveProbe Then
        If EventCount < 5 Then Target.Offset(0, 1).Value = "echo"
        Exit Sub
    End If
    If Target.Address = "$A$1" Then
        If SafeWriteMode Then Application.EnableEvents = False
        RevisarSiEsNumero Target
        If SafeWriteMode Then Application.EnableEvents = True
    End If
End Sub
Sub RevisarSiEsNumero(Target As Range)
    If IsNumeric(Target.Value) Then
        Target.Offset(0, 1).Value = "Numero"
    Else
        Target.Offset(0, 1).Value = "No es numero"
    End If
End Sub
